cmake_minimum_required(VERSION 3.20)
project(VoxelLandscape VERSION 1.0 LANGUAGES C CXX)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(VOXEL_BUILD_APP "Build the OpenGL application (downloads GLFW and Dear ImGui)" ON)

add_library(voxel_core src/world.cpp src/mesh.cpp)
find_package(Threads REQUIRED)
target_link_libraries(voxel_core PUBLIC Threads::Threads)
target_include_directories(voxel_core PUBLIC src)
if(MSVC)
    target_compile_options(voxel_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(voxel_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(voxel_tests tests/core_tests.cpp)
target_link_libraries(voxel_tests PRIVATE voxel_core)
add_test(NAME voxel_core COMMAND voxel_tests)
if(MINGW)
    target_link_options(voxel_tests PRIVATE -static-libgcc -static-libstdc++)
endif()

if(VOXEL_BUILD_APP)
    include(FetchContent)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz
        URL_HASH SHA256=c038d34200234d071fae9345bc455e4a8f2f544ab60150765d7704e08f3dac01)
    FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.91.9b.tar.gz
        URL_HASH SHA256=8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb)
    FetchContent_MakeAvailable(glfw imgui)
    find_package(OpenGL REQUIRED)
    add_library(voxel_ui STATIC
        ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
        ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
    target_include_directories(voxel_ui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
    target_link_libraries(voxel_ui PUBLIC glfw OpenGL::GL)
    add_executable(voxel src/main.cpp src/renderer.cpp)
    target_link_libraries(voxel PRIVATE voxel_core voxel_ui OpenGL::GL)
    if(MSVC)
        target_compile_options(voxel PRIVATE /W4 /utf-8)
    else()
        target_compile_options(voxel PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(MINGW)
        target_link_options(voxel PRIVATE -static-libgcc -static-libstdc++)
    endif()
endif()
